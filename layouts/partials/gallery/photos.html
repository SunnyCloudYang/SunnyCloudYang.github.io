{{- /* gallery/photos.html — enumerate a gallery album's bundle images and auto-generate everything the gallery needs: thumbnail (Resize 800x q80), LQIP blur placeholder (Resize 32x q20), and EXIF (camera/lens/aperture/ shutter/iso/focal) + shooting date, read straight from the image. An optional meta.yaml sidecar overrides title/caption/tags per filename. Returns a slice of photo dicts (src/thumb/blur/title/caption/date/tags/ exif), isomorphic to the old .Params.photos entries, so single.html and list.html consume the markup unchanged. */ -}} {{- $photos := slice -}} {{- $images := slice -}} {{- range .Resources -}} {{- $ext := lower (path.Ext .Name) -}} {{- if or (eq $ext ".jpg") (eq $ext ".jpeg") (eq $ext ".png") (eq $ext ".webp") -}} {{- $images = $images | append . -}} {{- end -}} {{- end -}} {{- $images = sort $images "Name" -}} {{- /* optional meta.yaml sidecar, keyed by resource .Name (e.g. "01.jpg") */ -}} {{- $meta := dict -}} {{- with .Resources.Get "meta.yaml" -}} {{- $meta = . | transform.Unmarshal -}} {{- end -}} {{- range $img := $images -}} {{- $name := $img.Name -}} {{- $m := index $meta $name | default dict -}} {{- $thumb := $img.Resize "800x q80" -}} {{- $blur := $img.Resize "32x q20" -}} {{- /* EXIF. Rational tags (FNumber/FocalLength) print as "N" or "N/D"; turn the latter into a float so f/5.6 (stored 28/5) renders as f/5.6, not f/28/5. */ -}} {{- $camera := "" -}}{{- $lens := "" -}}{{- $aperture := "" -}}{{- $shutter := "" -}}{{- $iso := "" -}}{{- $focal := "" -}}{{- $date := "" -}} {{- with $img.Exif -}} {{- $t := .Tags -}} {{- with $t.Model }}{{- $camera = printf "%v" . -}}{{- end -}} {{- with $t.LensModel }}{{- $lens = printf "%v" . -}}{{- end -}} {{- with $t.FNumber -}} {{- $s := printf "%v" . -}} {{- if in $s "/" -}}{{- $p := split $s "/" -}}{{- $aperture = printf "f/%v" (div (float (index $p 0)) (float (index $p 1))) -}} {{- else -}}{{- $aperture = printf "f/%v" (float $s) -}}{{- end -}} {{- end -}} {{- with $t.ExposureTime }}{{- $shutter = printf "%v" . -}}{{- end -}} {{- with $t.ISO }}{{- $iso = printf "%v" . -}}{{- end -}} {{- with $t.FocalLength -}} {{- $s := printf "%v" . -}} {{- if in $s "/" -}}{{- $p := split $s "/" -}}{{- $focal = printf "%vmm" (div (float (index $p 0)) (float (index $p 1))) -}} {{- else -}}{{- $focal = printf "%vmm" (float $s) -}}{{- end -}} {{- end -}} {{- with $t.DateTimeOriginal -}} {{- /* "2024:08:15 05:42:00" (string) -> take the date part, : to - */ -}} {{- $date = replace (index (split (printf "%v" .) " ") 0) ":" "-" -}} {{- end -}} {{- end -}} {{- /* title: sidecar > EXIF ImageDescription > filename */ -}} {{- $title := "" -}} {{- with index $m "title" }}{{- $title = . -}}{{- end -}} {{- if not $title -}}{{- with $img.Exif }}{{- with .Tags.ImageDescription }}{{- $title = printf "%v" . -}}{{- end -}}{{- end -}}{{- end -}} {{- if not $title }}{{- $title = $name -}}{{- end -}} {{- $caption := "" -}} {{- with index $m "caption" }}{{- $caption = . -}}{{- end -}} {{- $tags := slice -}} {{- with index $m "tags" }}{{- $tags = . -}}{{- end -}} {{- $exif := dict "camera" $camera "lens" $lens "aperture" $aperture "shutter" $shutter "iso" $iso "focal" $focal -}} {{- $photos = $photos | append (dict "src" $img.RelPermalink "thumb" $thumb.RelPermalink "blur" $blur.RelPermalink "title" $title "caption" $caption "date" $date "tags" $tags "exif" $exif) -}} {{- end -}} {{- return $photos -}}
